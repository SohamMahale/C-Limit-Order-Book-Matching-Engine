cmake_minimum_required(VERSION 3.14)
project(lob CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-Wall -Wextra)
include_directories(include)
add_executable(lob_cli src/main.cpp)
add_executable(lob_tests tests/test_book.cpp)
add_executable(lob_bench bench/bench.cpp)
add_executable(lob_research research/imbalance.cpp)
enable_testing()
add_test(NAME unit COMMAND lob_tests)
